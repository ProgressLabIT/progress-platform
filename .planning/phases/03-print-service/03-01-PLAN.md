```python
from datetime import datetime
from enum import Enum
from pydantic import BaseModel, Field
from models.base_models import ArangoDocument


class PrintFormat(str, Enum):
    ZPL = "zpl"
    PDF = "pdf"


class PrintJobRequest(BaseModel):
    printer_host: str
    printer_port: int = 9100
    format: PrintFormat
    data: str  # ZPL text or base64-encoded PDF
    copies: int = 1
    timeout_seconds: float = 5.0


class PrintJobRecord(ArangoDocument):
    status: str = "pending"  # pending | sent | failed
    printer_host: str
    printer_port: int
    format: PrintFormat
    copies: int
    timeout_seconds: float
    created_at: str | None = None  # ISO datetime string
    completed_at: str | None = None
    error: str | None = None
    error_detail: str | None = None


class PrintJobResult(BaseModel):
    ok: bool
    error: str | None = None  # connection_refused | timeout | send_error
    detail: str | None = None
```

Note: `data` is intentionally NOT stored in the DB record (it can be large base64 PDF). It is only passed through the SSE event payload.

**Endpoint 1: POST /print-job**
```python
@router.post('/print-job', dependencies=[Depends(auth.verify_token)])
async def create_print_job(job: PrintJobRequest):
    record = {
        "status": "pending",
        "printer_host": job.printer_host,
        "printer_port": job.printer_port,
        "format": job.format.value,
        "copies": job.copies,
        "timeout_seconds": job.timeout_seconds,
        "created_at": datetime.now(timezone.utc).isoformat(),
        "completed_at": None,
        "error": None,
        "error_detail": None,
    }
    result = db.collection("PrintJob").insert(record)
    job_key = result["_key"]

    sse_payload = job.model_dump()
    sse_payload["format"] = job.format.value
    sse_payload["job_id"] = job_key
    sse_payload["subtopic"] = "print-jobs"
    ServerEventManager.getInstance().enqueue(json.dumps(sse_payload))

    return {"job_id": job_key}
```

**Endpoint 2: GET /print-jobs/stream (SSE channel)**
```python
@router.get('/print-jobs/stream', dependencies=[Depends(auth.verify_token)])
async def print_job_stream(request: Request):
    return EventSourceResponse(
        ServerEventManager.getInstance().push_events(request, "print-jobs")
    )
```

**Endpoint 3: POST /print-jobs/{job_id}/result**
```python
@router.post('/print-jobs/{job_id}/result', dependencies=[Depends(auth.verify_token)])
async def print_job_result(job_id: str, result: PrintJobResult):
    now = datetime.now(timezone.utc).isoformat()
    update = {
        "_key": job_id,
        "status": "sent" if result.ok else "failed",
        "completed_at": now,
    }
    if not result.ok:
        update["error"] = result.error
        update["error_detail"] = result.detail
    db.collection("PrintJob").update(update)

    NotificationManager.getInstance().notifyGlobalRefresh()

    return {"ok": True}
```

CRITICAL details:
- The SSE payload MUST include `"subtopic": "print-jobs"` — this is how `ServerEventManager.enqueue()` routes events to the correct queue (it does `json.loads(message)['subtopic']`).
- The `format` field in the SSE payload must be the string value (`"zpl"` or `"pdf"`), not the Enum object — use `job.format.value`.
- The `data` field IS included in the SSE payload (so the print service receives it) but is NOT stored in the DB record (it can be large).
- The SSE stream endpoint follows the exact same pattern as `notification.py` — `EventSourceResponse(push_events(request, topic))`.
- All three endpoints use `dependencies=[Depends(auth.verify_token)]` — same pattern as existing print template endpoints.
- `notifyGlobalRefresh()` is the existing shorthand that sends a REFRESH notification to all browser WebSocket connections via Kafka.
