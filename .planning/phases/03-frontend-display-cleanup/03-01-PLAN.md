Add the `formatFloat` function and export it as a global property in `boot/filters.js`:

```javascript
/**
 * Format float value for display with clean precision.
 *
 * Prevents floating-point noise in the UI by rounding to specified decimal places
 * and optionally stripping trailing zeros for cleaner display.
 *
 * @param {number} value - Float value to format
 * @param {number} maxDecimals - Maximum decimal places (default: 6, matching backend standard)
 * @param {boolean} stripTrailingZeros - Remove trailing zeros (default: true)
 * @returns {string|number} Formatted value or '-' for null/NaN
 *
 * @example
 * formatFloat(99.999999)           // 100
 * formatFloat(100.000001)          // 100
 * formatFloat(1/3)                 // 0.333333
 * formatFloat(1/3, 2)              // 0.33
 * formatFloat(50.5000, 6, false)   // 50.500000
 * formatFloat(null)                // '-'
 */
export function formatFloat(value, maxDecimals = 6, stripTrailingZeros = true) {
  // Handle null/undefined
  if (value === null || value === undefined) {
    return '-';
  }

  // Handle non-numeric values
  if (typeof value !== 'number') {
    return value;
  }

  // Handle NaN
  if (isNaN(value)) {
    return '-';
  }

  // Handle Infinity
  if (!isFinite(value)) {
    return value; // Return Infinity or -Infinity as-is
  }

  // Round to maxDecimals using the same approach as backend
  const rounded = Math.round(value * 10 ** maxDecimals) / 10 ** maxDecimals;

  if (stripTrailingZeros) {
    // Convert to string with fixed decimals, then parse back to remove trailing zeros
    // This ensures "100.000000" becomes 100, "50.50" becomes 50.5, etc.
    return parseFloat(rounded.toFixed(maxDecimals));
  }

  return rounded;
}
```
