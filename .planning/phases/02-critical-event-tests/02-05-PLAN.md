               qt_planned=1.0, qt_confirmed=1.0, **overrides):
    info = {
        "event_type": "MOVEMENT_COMPLETED",
        "primary": True,
        "user_key": user_key,
        "movement_type": movement_type,
        "qt_planned": qt_planned,
        "qt_confirmed": qt_confirmed,
    }

    if position_from is not None:
