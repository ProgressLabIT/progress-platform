<!-- Exact pattern to follow — from webapps/main/src/lib/print/plugins/linkedText.js -->
import { text } from '@pdfme/schemas';
import { createLinkSchema, linkDefaults } from './linkConfig.js';

export function createLinkedText(customFields = []) {
  return {
    pdf: text.pdf,
    ui: text.ui,
    icon: textIcon,
    propPanel: {
      schema: (props) => ({
        ...createLinkSchema(customFields, props.activeSchema),
        ...(typeof text.propPanel.schema === 'function'
          ? text.propPanel.schema(props)
          : text.propPanel.schema),
      }),
      widgets: text.propPanel.widgets || {},
      defaultSchema: {
        ...text.propPanel.defaultSchema,
        ...linkDefaults,
      },
    },
  };
}

<!-- linkDefaults shape (from linkConfig.js) -->
export const linkDefaults = {
  linkType: 'none',
  linkValue: '',
  extraPath: '',
};

<!-- buildPlugins() current shape (from plugins/index.js) -->
export function buildPlugins(customFields = []) {
  const barcodes = createLinkedBarcodes(customFields);
  return {
    text: createLinkedText(customFields),
    image: createLinkedImage(customFields),
    qrcode: barcodes.qrcode,
    code128: barcodes.code128,
    code39: barcodes.code39,
    ean13: barcodes.ean13,
    gs1datamatrix: barcodes.gs1datamatrix,
  };
}
